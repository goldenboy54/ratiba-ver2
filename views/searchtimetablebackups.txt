 <script>
$(document).ready(function() {
  $('.select2').select2();

  // Auto-submit semester form
  $('#semester').on('change', function(){ $('#semesterForm').submit(); });

  // Auto hot search for filters
  $('#filterForm select').on('change', function(){ $('#filterForm').submit(); });

  // Function to get current academic year
  function getAcademicYear() {
    const now = new Date();
    const month = now.getMonth() + 1; // Jan = 0
    const year = now.getFullYear();
    return month >= 8 ? `${year}/${year+1}` : `${year-1}/${year}`;
  }

  // Browser print
  $('#frontendPrintBtn').on('click', function(){
    const now = new Date();
    const datetime = `${now.getFullYear()}-${now.getMonth()+1}-${now.getDate()}_${now.getHours()}-${now.getMinutes()}-${now.getSeconds()}`;
    const content = document.getElementById('timetableContent').innerHTML;
    const win = window.open('', '', 'width=1000,height=800');

    const academicYear = getAcademicYear();

    win.document.write('<html><head><title>ATC_Timetable_'+datetime+'</title>');
    win.document.write('<style>body{font-family:Arial,sans-serif;margin:20px;} table{width:100%;border-collapse:collapse;} th,td{border:1px solid #000;padding:6px;text-align:center;} th{background:#007bff;color:#fff;} tbody tr:nth-child(even){background:#f2f2f2;}</style>');
    win.document.write('</head><body>');
    win.document.write('<h1 style="text-align:center;">ARUSHA TECHNICAL COLLEGE</h1>');
    win.document.write('<h2 style="text-align:center;">Timetable - Semester <%= semester %> (' + academicYear + ')</h2>');
    win.document.write(content);
    win.document.write('</body></html>');
    win.document.close();
    win.focus();
    win.print();
  });

  // Frontend PDF download
  $('#frontendPdfBtn').on('click', function(){
    const now = new Date();
    const datetime = `${now.getFullYear()}-${now.getMonth()+1}-${now.getDate()}_${now.getHours()}-${now.getMinutes()}-${now.getSeconds()}`;
    const filename = `ATC_Timetable_${datetime}.pdf`;

    const { jsPDF } = window.jspdf;
    const doc = new jsPDF('l', 'pt', 'a4');
    const element = document.getElementById('timetableContent');

    const academicYear = getAcademicYear();

    html2canvas(element, { scale: 2 }).then(canvas => {
      const imgData = canvas.toDataURL('image/png');
      const imgProps = doc.getImageProperties(imgData);
      const pdfWidth = doc.internal.pageSize.getWidth();
      const pdfHeight = doc.internal.pageSize.getHeight();
      const imgWidth = pdfWidth;
      const imgHeight = (imgProps.height * imgWidth) / imgProps.width;

      let heightLeft = imgHeight;
      let position = 50;

      // Add header
      doc.setFontSize(16);
      doc.text("ARUSHA TECHNICAL COLLEGE", pdfWidth/2, 20, { align: "center" });
      doc.text(`Timetable - Semester <%= semester %> (${academicYear})`, pdfWidth/2, 40, { align: "center" });

      doc.addImage(imgData, 'PNG', 0, position, imgWidth, imgHeight);
      heightLeft -= (pdfHeight - position);

      while (heightLeft > 0) {
        position = heightLeft - imgHeight + 50;
        doc.addPage();
        doc.addImage(imgData, 'PNG', 0, position, imgWidth, imgHeight);
        heightLeft -= (pdfHeight - 50);
      }

      doc.save(filename);
    });
  });
});
</script>
